Python Script Modified for Logarithmic Function

 
# Code is modified to feed a logarithmic function 
# to the Newton-Raphson code.

def f(x):
    return (np.log(x))/(np.log(3)) + 10

def df(x):
    return 1 / (x * (np.log(3)))

def ddf(x):
    return 1 / ((x * x) * (np.log(3)))

Python Script Modified for Rational Function

 
# Code is modified to feed a rational function

Python Script Modified for Radical Root Function

 
# Code is modified to feed a radical root 
# function to the Newton-Raphson code.

def f(x):
    return np.cbrt(x)

def df(x):
    return 1/ (3 * (np.cbrt(x * x)))

def ddf(x):
    return -2 / (9 * (np.cbrt(x * x * x * x * x)))
